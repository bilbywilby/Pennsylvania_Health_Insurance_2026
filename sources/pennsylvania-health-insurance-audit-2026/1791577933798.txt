"""
SYSTEM HEALTH MONITORING
==========================
Monitors the audit pipeline itself to prevent silent failures.
Alerts on:
- Pipeline downtime
- Data source unavailability (>5 min)
- Schema validation failures
- Storage capacity approaching limit
"""

import os
import time
import psutil
import sqlite3
import shutil
from pathlib import Path
from datetime import datetime, timedelta
import logging
from typing import Dict, Any, List, Optional
import json

logger = logging.getLogger(__name__)

class PipelineHealthMonitor:
    """
    Checks that the audit engine remains operational.

    Integration Points:
        - Run via cron every 15 minutes
        - Post alerts to same channels as audit notifications
        - Export Prometheus metrics for Grafana dashboards
    """

    def __init__(self, db_path: Path):
        self.db_path = db_path
        self.db = None
        self.data_root = Path("./data")
        self.alert_history: List[Dict[str, Any]] = []

    def connect(self) -> None:
        """Establish database connection with timeout."""
        try:
            self.db = sqlite3.connect(str(self.db_path), timeout=10)
            self.db.row_factory = sqlite3.Row
        except sqlite3.Error as e:
            logger.error(f"Database connection failed: {e}")
            raise

    def disconnect(self) -> None:
        """Gracefully close database connection."""
        if self.db:
            self.db.close()

    def check_pipeline_uptime(self) -> Dict[str, Any]:
        """Check if last successful audit exists within 30 days."""
        if not self.db:
            self.connect()

        cursor = self.db.execute(
            "SELECT MAX(audit_date) as last_run, overall_status FROM audits"
        )
        row = cursor.fetchone()

        if row['last_run'] is None:
            return {
                "status": "CRITICAL",
                "message": "No audits found in database",
                "days_since_run": None,
                "action_required": True
            }

        last_run_date = datetime.strptime(row['last_run'], '%Y-%m-%d')
        days_since_run = (datetime.now() - last_run_date).days

        if days_since_run > 30:
            return {
                "status": "CRITICAL",
                "message": f"No audit run in {days_since_run} days",
                "days_since_run": days_since_run,
                "action_required": True
            }
        elif days_since_run > 14:
            return {
                "status": "WARNING",
                "message": f"Last audit {days_since_run} days ago",
                "days_since_run": days_since_run,
                "action_required": False
            }

        return {
            "status": "HEALTHY",
            "message": f"Last audit {days_since_run} days ago",
            "days_since_run": days_since_run,
            "last_run": row['last_run'],
            "action_required": False
        }

    def check_storage(self) -> Dict[str, Any]:
        """Check disk space usage with warning thresholds."""
        total, used, free = shutil.disk_usage(str(self.data_root))

        usage_percent = (used / total) * 100
        warning_threshold = 85
        critical_threshold = 95

        if usage_percent >= critical_threshold:
            status = "CRITICAL"
        elif usage_percent >= warning_threshold:
            status = "WARNING"
        else:
            status = "HEALTHY"

        return {
            "status": status,
            "total_gb": f"{total / (1024**3):.2f}",
            "used_gb": f"{used / (1024**3):.2f}",
            "free_gb": f"{free / (1024**3):.2f}",
            "usage_percent": round(usage_percent, 2),
            "warning_threshold": warning_threshold,
            "critical_threshold": critical_threshold,
            "action_required": usage_percent >= warning_threshold
        }

    def check_data_source_connectivity(self, sources: List[str]) -> Dict[str, Any]:
        """Test connectivity to all primary data sources."""
        from http_client import safe_get

        results = {}
        failures = []

        async def test_source(url: str, name: str) -> Dict[str, Any]:
            try:
                content = await safe_get(url)
                return {"name": name, "status": "OK", "latency_ms": len(content) / 100}
            except Exception as e:
                return {"name": name, "status": "FAILED", "error": str(e)}

        # Run all checks concurrently
        import asyncio
        tasks = [test_source(s['url'], s['name']) for s in sources]
        results_list = asyncio.run(asyncio.gather(*tasks, return_exceptions=True))

        for result in results_list:
            if isinstance(result, Exception):
                failures.append(result)
            else:
                results[result['name']] = result

        if failures:
            return {
                "status": "WARNING",
                "successful_checks": len([r for r in results_list if not isinstance(r, Exception)]),
                "failed_checks": len(failures),
                "details": results,
                "action_required": len(failures) > 0
            }

        return {
            "status": "HEALTHY",
            "checks_passed": len(results),
            "details": results,
            "action_required": False
        }

    def check_schema_validation_failures(self) -> Dict[str, Any]:
        """Scan recent audit reports for schema violations."""
        reports_dir = Path("./reports")
        if not reports_dir.exists():
            return {"status": "UNKNOWN", "message": "Reports directory not found"}

        validation_errors = []

        for report_file in reports_dir.glob("audit_*.json"):
            try:
                from schemas import AuditReport
                with open(report_file) as f:
                    data = json.load(f)
                AuditReport(**data)  # Validate
            except Exception as e:
                validation_errors.append({
                    "file": str(report_file),
                    "error": str(e),
                    "timestamp": datetime.now().isoformat()
                })

        if validation_errors:
            return {
                "status": "CRITICAL",
                "validation_failures": len(validation_errors),
                "details": validation_errors,
                "action_required": True
            }

        return {
            "status": "HEALTHY",
            "validation_failures": 0,
            "action_required": False
        }

    def run_full_health_check(self) -> Dict[str, Any]:
        """Execute all health checks and compile report."""
        checks = {
            "pipeline_uptime": self.check_pipeline_uptime(),
            "storage": self.check_storage(),
            "data_source_connectivity": self.check_data_source_connectivity([
                {"name": "PID", "url": settings.pid_base_url},
                {"name": "CMS IDR", "url": settings.cms_base_url},
                {"name": "Pennie", "url": settings.pennie_base_url}
            ]),
            "schema_validation": self.check_schema_validation_failures()
        }

        # Determine overall system health
        statuses = [c['status'] for c in checks.values()]

        if "CRITICAL" in statuses:
            overall_status = "CRITICAL"
        elif "WARNING" in statuses:
            overall_status = "WARNING"
        else:
            overall_status = "HEALTHY"

        report = {
            "overall_status": overall_status,
            "timestamp": datetime.now().isoformat(),
            "checks": checks,
            "action_items": [
                f"[{k}] {v['message']}" for k, v in checks.items()
                if v.get('action_required', False)
            ]
        }

        # Log to file
        log_path = Path("./logs/system_health.json")
        log_path.parent.mkdir(parents=True, exist_ok=True)

        with open(log_path, 'a') as f:
            f.write(json.dumps(report) + "\n")

        # Alert if critical
        if overall_status in ["CRITICAL", "WARNING"]:
            self._send_health_alert(report)

        return report

    def _send_health_alert(self, health_report: Dict[str, Any]):
        """Send alert for critical system health issues."""
        if not settings.slack_webhook_url:
            logger.warning("Slack webhook not configured, skipping health alert")
            return

        payload = {
            "text": f"🚨 SYSTEM HEALTH ALERT\nOverall Status: {health_report['overall_status']}",
            "blocks": [
                {
                    "type": "section",
                    "text": {
                        "type": "mrkdwn",
                        "text": "\n".join(health_report.get("action_items", []))
                    }
                }
            ]
        }

        import aiohttp
        asyncio.create_task(self._post_webhook(settings.slack_webhook_url, payload))

    async def _post_webhook(self, url: str, payload: dict) -> None:
        """Post to webhook URL with error handling."""
        async with aiohttp.ClientSession() as session:
            try:
                async with session.post(url, json=payload, timeout=10) as resp:
                    logger.info(f"Health alert posted: {resp.status}")
            except Exception as e:
                logger.error(f"Failed to post health alert: {e}")

    """Background thread for periodic maintenance."""

            # Run health check every 15 minutes
            health_report = health_monitor.run_full_health_check()

            # If WARNING or CRITICAL, trigger immediate audit
            if health_report['overall_status'] in ['WARNING', 'CRITICAL']:
                logger.warning("Scheduled health check triggered manual audit")
                # pipeline.run_full_audit() would be called here

            time.sleep(900)  # 15 minutes
        except Exception as e:
            logger.error(f"Scheduled task error: {e}")
            time.sleep(60)

# Signal handlers for graceful shutdown
